========== удаление вопроса с тремя вариантами ==========
BEGIN (implicit)
SELECT questions.id, questions.survey_id, questions.question_type, questions.position, questions.required, questions.body 
FROM questions 
WHERE questions.id = $1::UUID
SELECT answer_options.id, answer_options.question_id, answer_options.position, answer_options.text 
FROM answer_options 
WHERE $1::UUID = answer_options.question_id
DELETE FROM answer_options WHERE answer_options.id = $1::UUID
DELETE FROM questions WHERE questions.id = $1::UUID
COMMIT
========== конец ==========
вариантов ответа осталось: 0
